# create date, year, and month column
date_processing = function(data, col, out_name){
  dates = as.Date(data[[col]])
  years = lubridate::year(dates)
  months = format(dates, "%m-%Y")
  column_names = paste0(out_name, c("_date", "_year", "months"))
  out = data.frame(dates, years, months)
  names(out) = column_names
  return(out)
}

wildfires = cbind(wildfires, date_processing(wildfires, col = "DISCOVERY_DATE", "discovery_processed"))

Takeaways from missing data

Some columns have too much missingness to ignore or reliably impute. Those are:

  • LOCAL_FIRE_REPORT_ID
  • LOCAL_INCIDENT_ID
  • FIRE_CODE
  • FIRE_NAME
  • ICS_209_INCIDENT_NUMBER
  • ICS_209_NAME
  • MTBS_ID
  • MTBS_FIRE_NAME
  • COMPLEX_NAME
  • DISCOVERY_TIME

COUNTY is the less nicely formatted version of FIPS_CODE, so we can remove the COUNTY column without losing any information.

The CONT… columns have a large number of NA values, but NA values may indicate that they are not multi-day fires. The reason for these NA’s is not made clear in the data source site. I will keep these columns, but limit to fires with complete informtion when making use of these columns.

  • CONT_DATE

  • CONT_DOY

  • CONT_TIME

COUNTY, FIPS_CODE, and FIPS_NAME are missing in over one third of entries, but they may be determined using lat/lon coordinates.

  • COUNTY (but we’re going to remove this anyway)

  • FIPS_CODE

  • FIPS_NAME

wildfires <- wildfires %>% select(-COUNTY)

# get US county geometries
US_counties <- tigris::counties()

# select lat/lon coordinates and format as an sf points object
lat_lon_coords = wildfires %>% select(LATITUDE, LONGITUDE)
latlon_pts <- sf::st_as_sf(lat_lon_coords, coords = c("LONGITUDE", "LATITUDE"), crs = 4326)

# transform counties and points geometries to have the same coordinate reference system
US_counties <- sf::st_transform(US_counties, crs = sf::st_crs(latlon_pts))

# find the index of the county each point falls in 
intersects <- sf::st_intersects(latlon_pts, US_counties)

# there were two points that were matched to two counties. In those cases, I just selected the first one. 
county_indexes <- lapply(intersects, function(x){x[1]}) %>% unlist()

# identify the name and fips code of the matched county
latlon_counties <- US_counties %>% 
  select("FIPS_NAME_calculated" = "NAME", 
         "FIPS_CODE_calculated" = "COUNTYFP") %>%
  as.data.frame() %>%
  select(-geometry)

point_counties <- latlon_counties[county_indexes,]

# join information back with the wildfire dataset and only use calculated values if original values are missing
wildfires <- cbind(wildfires, point_counties)
wildfires <- wildfires %>% mutate(FIPS_NAME = ifelse(is.na(FIPS_NAME), FIPS_NAME_calculated, FIPS_NAME),
                                  FIPS_CODE = ifelse(is.na(FIPS_CODE), FIPS_CODE_calculated, FIPS_CODE)) %>%
  select(-c(FIPS_NAME_calculated, FIPS_CODE_calculated))
row.names(wildfires) <- NULL

# add lat/lon points since that could be useful
wildfires <- cbind(wildfires, latlon_pts %>% rename("latlon_geometry" = "geometry"))

                       column_nm,

      color = "black"
